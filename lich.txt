    let currentViewDate = new Date();

    // Thuật toán tính ca trực cho một ngày bất kỳ
    function getShiftForDate(date) {
        const baseDate = new Date(2026, 9, 4); // 04/10/2026 là Chủ Nhật WW41
        const targetDate = new Date(date.getFullYear(), date.getMonth(), date.getDate());
        
        const diffTime = targetDate - baseDate;
        const diffDays = Math.floor(diffTime / (1000 * 60 * 60 * 24));
        const weekDiff = Math.floor(diffDays / 7);
        const dayOfWeek = (targetDate.getDay()); // 0 = CN, 1 = T2, ..., 6 = T7

        // Ngoại lệ riêng cho tuần WW 40 (27/09/2026 - 03/10/2026)
        const ww40Start = new Date(2026, 8, 27);
        const ww40End = new Date(2026, 9, 3);
        if (targetDate >= ww40Start && targetDate <= ww40End) {
            const ww40Map = {
                0: { shift: "D1", block: "A123" },
                1: { shift: "D3", block: "A123" },
                2: { shift: "AC4", block: "D123" },
                3: { shift: "AC4", block: "D123" },
                4: { shift: "AC4", block: "D123" },
                5: { shift: "D2", block: "A123" },
                6: { shift: "AC4", block: "A123" }
            };
            return ww40Map[dayOfWeek];
        }

        // Chu kỳ 6 tuần
        let cycleWeek = (weekDiff % 6);
        if (cycleWeek < 0) cycleWeek += 6;

        let shift = "";
        let block = "";

        if (cycleWeek === 0 || cycleWeek === 1 || cycleWeek === 2) { // Tuần 1, 2, 3 (Primary A)
            block = (dayOfWeek >= 0 && dayOfWeek <= 2) ? "A123" : "D123";
            if (dayOfWeek === 0) shift = "A1";
            else if (dayOfWeek === 1) shift = "A2";
            else if (dayOfWeek === 2) shift = "A3";
            else shift = "AC4";
        } else if (cycleWeek === 3) { // Tuần 4 (Transition)
            block = (dayOfWeek >= 0 && dayOfWeek <= 1) ? "D123" : "B123";
            if (dayOfWeek === 0) shift = "A2";
            else if (dayOfWeek === 1) shift = "A3";
            else if (dayOfWeek === 2) shift = "B1";
            else shift = "BD4";
        } else { // Tuần 5, 6 (Primary B)
            block = "B123";
            if (dayOfWeek === 0) shift = "B2";
            else if (dayOfWeek === 1) shift = "B3";
            else if (dayOfWeek === 2) shift = "B1";
            else shift = "BD4";
        }

        return { shift, block };
    }

    // Kiểm tra xem một ngày có phải ca đi làm của người dùng không
    function isUserWorkingOnDate(date, selectedShift) {
        const info = getShiftForDate(date);
        if (selectedShift === "A123") {
            return info.block === "A123";
        }
        return info.shift === selectedShift;
    }

    function updateShiftList() {
        const dept = document.getElementById("deptSelect").value;
        const shiftSelect = document.getElementById("shiftSelect");
        shiftSelect.innerHTML = "";

        const options = SHIFT_OPTIONS[dept] || SHIFT_OPTIONS["OS_ES_DAY"];
        options.forEach(opt => {
            const el = document.createElement("option");
            el.value = opt.code;
            el.textContent = opt.label;
            shiftSelect.appendChild(el);
        });

        // Khôi phục giá trị đã lưu nếu có
        const savedShift = localStorage.getItem("selectedShift");
        if (savedShift) {
            shiftSelect.value = savedShift;
        }

        saveConfigAndRefresh();
    }

    function saveConfigAndRefresh() {
        const dept = document.getElementById("deptSelect").value;
        const shift = document.getElementById("shiftSelect").value;
        localStorage.setItem("selectedDept", dept);
        localStorage.setItem("selectedShift", shift);

        checkScheduleToday();
        renderCalendar();
    }

    function checkScheduleToday() {
        const today = new Date();
        const selectedShift = document.getElementById("shiftSelect").value;
        const isWorking = isUserWorkingOnDate(today, selectedShift);
        const info = getShiftForDate(today);

        const badge = document.getElementById("statusBadge");
        const text = document.getElementById("statusText");
        const desc = document.getElementById("statusDesc");

        const dateStr = `${today.getDate().toString().padStart(2, '0')}/${(today.getMonth()+1).toString().padStart(2, '0')}/${today.getFullYear()}`;

        if (isWorking) {
            badge.className = "status-badge working";
            badge.textContent = "HÔM NAY CÓ CA TRỰC";
            text.textContent = `Bạn đi làm: Ca ${info.shift} (${info.block})`;
            text.style.color = "#DC2626";
            desc.textContent = `Ngày: ${dateStr}. Nhớ chuẩn bị đi làm đúng giờ nhé!`;
        } else {
            badge.className = "status-badge off";
            badge.textContent = "HÔM NAY ĐƯỢC NGHỈ";
            text.textContent = `Hôm nay là ca ${info.shift} (${info.block})`;
            text.style.color = "#16A34A";
            desc.textContent = `Ngày: ${dateStr}. Chúc bạn có một ngày nghỉ ngơi vui vẻ!`;
        }
    }

    function renderCalendar() {
        const grid = document.getElementById("calendarGrid");
        grid.innerHTML = "";

        const year = currentViewDate.getFullYear();
        const month = currentViewDate.getMonth();

        document.getElementById("calendarMonthTitle").textContent = `Tháng ${month + 1} / ${year}`;

        const dayNames = ["CN", "T2", "T3", "T4", "T5", "T6", "T7"];
        dayNames.forEach(name => {
            const el = document.createElement("div");
            el.className = "cal-day-name";
            el.textContent = name;
            grid.appendChild(el);
        });

        const firstDay = new Date(year, month, 1).getDay();
        const daysInMonth = new Date(year, month + 1, 0).getDate();
        const today = new Date();
        const selectedShift = document.getElementById("shiftSelect").value;

        for (let i = 0; i < firstDay; i++) {
            const empty = document.createElement("div");
            empty.className = "cal-cell empty";
            grid.appendChild(empty);
        }

        for (let day = 1; day <= daysInMonth; day++) {
            const thisDate = new Date(year, month, day);
            const isWork = isUserWorkingOnDate(thisDate, selectedShift);
            const shiftInfo = getShiftForDate(thisDate);
            const isToday = (today.getDate() === day && today.getMonth() === month && today.getFullYear() === year);

            const cell = document.createElement("div");
            cell.className = "cal-cell" + (isWork ? " work-day" : "") + (isToday ? " today" : "");

            cell.innerHTML = `
                <span>${day}</span>
                <span class="shift-tag">${shiftInfo.shift}</span>
            `;

            grid.appendChild(cell);
        }
    }

    function changeMonth(delta) {
        currentViewDate.setMonth(currentViewDate.getMonth() + delta);
        renderCalendar();
    }

    function requestNotification() {
        if (!("Notification" in window)) {
            alert("Trình duyệt không hỗ trợ thông báo tự động.");
            return;
        }

        Notification.requestPermission().then(permission => {
            if (permission === "granted") {
                new Notification("Lịch Làm Việc", {
                    body: "Đã bật nhắc nhở thành công! Ứng dụng sẽ thông báo mỗi khi đến ngày trực ca.",
                    icon: "https://vnatmfg.github.io/web-app/favicon.ico"
                });
            } else {
                alert("Bạn đã từ chối quyền thông báo trên trình duyệt.");
            }
        });
    }

    // Khởi tạo ứng dụng khi mở trang
    window.onload = function() {
        const savedDept = localStorage.getItem("selectedDept") || "OS_ES_DAY";
        document.getElementById("deptSelect").value = savedDept;
        updateShiftList();
    };
